foreach(check tls_io tls_options plaintext_io)
  add_executable(eli_${check}_test ${check}.c ../../lua/onelua.c)
  target_compile_definitions(eli_${check}_test PRIVATE MAKE_LIB)
  get_target_property(lss_test_includes lssocket INCLUDE_DIRECTORIES)
  get_target_property(http_test_includes lcorehttp INCLUDE_DIRECTORIES)
  target_include_directories(eli_${check}_test PRIVATE ${lss_test_includes} ${http_test_includes})
  get_target_property(lss_test_libraries eli LINK_LIBRARIES)
  target_link_libraries(eli_${check}_test PRIVATE ${lss_test_libraries})
  if(UNIX AND NOT APPLE)
    target_link_libraries(eli_${check}_test PRIVATE m dl)
  endif()
  add_test(NAME eli_${check} COMMAND eli_${check}_test)
  set_tests_properties(eli_${check} PROPERTIES TIMEOUT 15)
endforeach()
